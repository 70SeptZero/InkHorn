cmake_minimum_required(VERSION 3.25)
cmake_policy(SET CMP0141 NEW)
# Embedded debug information avoids a separate mspdbsrv process while retaining debugging.
set(CMAKE_MSVC_DEBUG_INFORMATION_FORMAT "$<$<CONFIG:Debug,RelWithDebInfo>:Embedded>")
project(InkHorn VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(INKHORN_WITH_PDFIUM "Build the actual PDF reader" ON)
option(INKHORN_DYNAMIC_PDFIUM "Development only: load a local PDFium DLL" OFF)
option(INKHORN_BUILD_TESTS "Build geometry and PDF roundtrip tests" ON)
set(PDFIUM_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/third_party/pdfium" CACHE PATH "Locally built PDFium SDK")
if(MSVC)
  # CMake may double-decode the UTF-8 /showIncludes prefix with a Chinese-only MSVC install.
  if(CMAKE_CXX_CL_SHOWINCLUDES_PREFIX MATCHES "^娉")
    set(CMAKE_CXX_CL_SHOWINCLUDES_PREFIX "注意: 包含文件: ")
    set(CMAKE_CL_SHOWINCLUDES_PREFIX "注意: 包含文件: ")
  endif()
  # Matches the non-component Chromium/PDFium build. Release CRT in all configs.
  set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreaded)
  add_compile_definitions(_ITERATOR_DEBUG_LEVEL=0)
  add_compile_options(/utf-8 /W4 /permissive-)
else()
  add_compile_options(-Wall -Wextra -Wpedantic)
endif()
if(MINGW)
  add_link_options(-static -static-libgcc -static-libstdc++)
endif()
add_library(inkhorn_core src/core.cpp)
target_include_directories(inkhorn_core PUBLIC src)
if(WIN32)
  enable_language(RC)
  add_compile_definitions(UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
  add_library(inkhorn_runtime src/app_paths.cpp src/runtime_trace.cpp)
  target_include_directories(inkhorn_runtime PUBLIC src)
  add_library(inkhorn_pdf src/pdf_engine.cpp)
  add_library(inkhorn_storage src/storage.cpp)
  target_link_libraries(inkhorn_storage PUBLIC inkhorn_core inkhorn_runtime)
  target_link_libraries(inkhorn_pdf PUBLIC inkhorn_core PRIVATE bcrypt inkhorn_runtime)
  target_compile_definitions(inkhorn_pdf PUBLIC INKHORN_WITH_PDFIUM=$<BOOL:${INKHORN_WITH_PDFIUM}>)
  if(INKHORN_WITH_PDFIUM)
    find_path(PDFIUM_INCLUDE_DIR fpdfview.h PATHS "${PDFIUM_ROOT}/include" NO_DEFAULT_PATH REQUIRED)
    target_include_directories(inkhorn_pdf PRIVATE "${PDFIUM_INCLUDE_DIR}")
    if(INKHORN_DYNAMIC_PDFIUM)
      target_compile_definitions(inkhorn_pdf PRIVATE INKHORN_DYNAMIC_PDFIUM=1)
    else()
      if(NOT MSVC)
        message(FATAL_ERROR "Static PDFium needs the MSVC/clang-cl ABI. In CLion select Visual Studio x64, not bundled MinGW.")
      endif()
      find_library(PDFIUM_LIBRARY NAMES pdfium PATHS "${PDFIUM_ROOT}/lib" NO_DEFAULT_PATH REQUIRED)
      target_link_libraries(inkhorn_pdf PRIVATE "${PDFIUM_LIBRARY}" advapi32 gdi32 user32 winmm dbghelp shlwapi)
    endif()
  endif()
  add_executable(InkHorn WIN32 src/main.cpp src/app.cpp src/ui.cpp src/input_diagnostics.cpp assets/app.rc)
  find_package(Python3 COMPONENTS Interpreter REQUIRED)
  file(GLOB INKHORN_SVG_INPUTS CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/assets/icons/*.svg")
  set(INKHORN_SVG_HEADER "${CMAKE_CURRENT_BINARY_DIR}/generated/generated_icons.hpp")
  add_custom_command(OUTPUT "${INKHORN_SVG_HEADER}"
    COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/scripts/Compile-Svg.py"
      "${CMAKE_CURRENT_SOURCE_DIR}/assets/icons" "${INKHORN_SVG_HEADER}"
    DEPENDS ${INKHORN_SVG_INPUTS} "${CMAKE_CURRENT_SOURCE_DIR}/scripts/Compile-Svg.py"
    VERBATIM)
  target_sources(InkHorn PRIVATE "${INKHORN_SVG_HEADER}")
  target_include_directories(InkHorn PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
  # New/replaced ICO files are embedded on the next CLion CMake reload/build.
  file(GLOB INKHORN_ICON_INPUTS CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/assets/icons/*.ico")
  set(INKHORN_ICON_NAMES select pen1 pen2 eraser highlighter shapes undo redo save)
  set(INKHORN_SVG_NAMES select pen pen eraser highlighter rounded-rectangle undo redo save)
  set(INKHORN_ICON_RC "#include <windows.h>\n#pragma code_page(65001)\n")
  set(INKHORN_ICON_ID 300)
  foreach(INKHORN_ICON_NAME IN LISTS INKHORN_ICON_NAMES)
    math(EXPR INKHORN_ICON_INDEX "${INKHORN_ICON_ID} - 300")
    list(GET INKHORN_SVG_NAMES ${INKHORN_ICON_INDEX} INKHORN_SVG_NAME)
    set(INKHORN_ICON_PATH "${CMAKE_CURRENT_SOURCE_DIR}/assets/icons/${INKHORN_ICON_NAME}.ico")
    if(EXISTS "${INKHORN_ICON_PATH}" AND NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/assets/icons/${INKHORN_SVG_NAME}.svg")
      string(APPEND INKHORN_ICON_RC "${INKHORN_ICON_ID} ICON \"${INKHORN_ICON_PATH}\"\n")
    endif()
    math(EXPR INKHORN_ICON_ID "${INKHORN_ICON_ID} + 1")
  endforeach()
  foreach(INKHORN_MASK_NAME pen1-mask pen2-mask highlighter-mask)
    if(INKHORN_MASK_NAME STREQUAL "pen1-mask")
      set(INKHORN_ICON_ID 311)
      set(INKHORN_SVG_NAME pen)
    elseif(INKHORN_MASK_NAME STREQUAL "pen2-mask")
      set(INKHORN_ICON_ID 312)
      set(INKHORN_SVG_NAME pen)
    elseif(INKHORN_MASK_NAME STREQUAL "highlighter-mask")
      set(INKHORN_ICON_ID 314)
      set(INKHORN_SVG_NAME highlighter)
    else()
      set(INKHORN_ICON_ID 100)
    endif()
    set(INKHORN_ICON_PATH "${CMAKE_CURRENT_SOURCE_DIR}/assets/icons/${INKHORN_MASK_NAME}.ico")
    if(EXISTS "${INKHORN_ICON_PATH}" AND NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/assets/icons/${INKHORN_SVG_NAME}.svg")
      string(APPEND INKHORN_ICON_RC "${INKHORN_ICON_ID} ICON \"${INKHORN_ICON_PATH}\"\n")
    endif()
  endforeach()
  foreach(INKHORN_EXTRA_NAME circle line arrow chevron-left)
    if(INKHORN_EXTRA_NAME STREQUAL "circle")
      set(INKHORN_ICON_ID 330)
    elseif(INKHORN_EXTRA_NAME STREQUAL "line")
      set(INKHORN_ICON_ID 331)
    elseif(INKHORN_EXTRA_NAME STREQUAL "arrow")
      set(INKHORN_ICON_ID 332)
    else()
      set(INKHORN_ICON_ID 333)
    endif()
    set(INKHORN_ICON_PATH "${CMAKE_CURRENT_SOURCE_DIR}/assets/icons/${INKHORN_EXTRA_NAME}.ico")
    if(EXISTS "${INKHORN_ICON_PATH}" AND NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/assets/icons/${INKHORN_EXTRA_NAME}.svg")
      string(APPEND INKHORN_ICON_RC "${INKHORN_ICON_ID} ICON \"${INKHORN_ICON_PATH}\"\n")
      list(APPEND INKHORN_ICON_INPUTS "${INKHORN_ICON_PATH}")
    endif()
  endforeach()
  foreach(INKHORN_APP_ICON "assets/logo/app-tohru.ico" "assets/logo/app.ico" "assets/icons/app.ico")
    if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/${INKHORN_APP_ICON}")
      string(APPEND INKHORN_ICON_RC "100 ICON \"${CMAKE_CURRENT_SOURCE_DIR}/${INKHORN_APP_ICON}\"\n")
      list(APPEND INKHORN_ICON_INPUTS "${CMAKE_CURRENT_SOURCE_DIR}/${INKHORN_APP_ICON}")
      break()
    endif()
  endforeach()
  file(CONFIGURE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/icons.rc" CONTENT "${INKHORN_ICON_RC}" @ONLY)
  if(INKHORN_ICON_RC MATCHES " ICON ")
    target_sources(InkHorn PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/icons.rc")
    set_source_files_properties("${CMAKE_CURRENT_BINARY_DIR}/icons.rc"
      PROPERTIES OBJECT_DEPENDS "${INKHORN_ICON_INPUTS}")
  endif()
  set(INKHORN_NOTICES_FILE "${CMAKE_CURRENT_SOURCE_DIR}/THIRD_PARTY_NOTICES.txt")
  if(EXISTS "${PDFIUM_ROOT}/THIRD_PARTY_NOTICES.txt")
    set(INKHORN_NOTICES_FILE "${PDFIUM_ROOT}/THIRD_PARTY_NOTICES.txt")
  endif()
  file(TO_CMAKE_PATH "${INKHORN_NOTICES_FILE}" INKHORN_NOTICES_FILE)
  configure_file(assets/notices.rc.in "${CMAKE_CURRENT_BINARY_DIR}/notices.rc" @ONLY)
  target_sources(InkHorn PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/notices.rc")
  if(MSVC)
    target_sources(InkHorn PRIVATE assets/app.manifest)
  else()
    target_compile_definitions(InkHorn PRIVATE INKHORN_MANIFEST_IN_RC)
  endif()
  target_link_libraries(InkHorn PRIVATE inkhorn_pdf inkhorn_storage d2d1 dwrite dwmapi comdlg32 comctl32 shell32 shcore ole32 uuid windowscodecs)
  target_include_directories(InkHorn PRIVATE assets)
  if(MINGW)
    target_link_options(InkHorn PRIVATE -municode -static-libgcc -static-libstdc++)
  endif()
  set_target_properties(InkHorn PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
  install(TARGETS InkHorn RUNTIME DESTINATION .)
  set(CPACK_GENERATOR ZIP)
  set(CPACK_PACKAGE_FILE_NAME "InkHorn-${PROJECT_VERSION}-windows-x64")
  include(CPack)
endif()
if(INKHORN_BUILD_TESTS)
  enable_testing()
  add_executable(core_tests tests/core_tests.cpp)
  target_link_libraries(core_tests PRIVATE inkhorn_core)
  add_test(NAME core_geometry_and_history COMMAND core_tests)
  if(WIN32)
    add_executable(storage_tests tests/storage_tests.cpp)
    target_link_libraries(storage_tests PRIVATE inkhorn_storage)
    add_test(NAME recovery_roundtrip COMMAND storage_tests)
    add_executable(input_tests tests/input_tests.cpp src/input_diagnostics.cpp)
    target_include_directories(input_tests PRIVATE src)
    target_link_libraries(input_tests PRIVATE user32)
    add_test(NAME input_diagnostic_roundtrip COMMAND input_tests)
    add_executable(runtime_tests tests/runtime_tests.cpp)
    target_link_libraries(runtime_tests PRIVATE inkhorn_runtime)
    add_test(NAME runtime_watchdog COMMAND runtime_tests)
    add_test(NAME native_ui_smoke COMMAND InkHorn --smoke-test)
    add_test(NAME handwriting_performance COMMAND InkHorn --performance-test)
    add_test(NAME continuous_short_strokes COMMAND InkHorn --short-stroke-test)
    add_test(NAME ink_tile_quality COMMAND InkHorn --ink-quality-test)
    add_test(NAME svg_vector_resources COMMAND "${Python3_EXECUTABLE}"
      "${CMAKE_CURRENT_SOURCE_DIR}/tests/svg_tests.py" "${CMAKE_CURRENT_SOURCE_DIR}/scripts/Compile-Svg.py")
    set_tests_properties(native_ui_smoke PROPERTIES TIMEOUT 30)
    set_tests_properties(handwriting_performance PROPERTIES TIMEOUT 45)
    set_tests_properties(continuous_short_strokes PROPERTIES TIMEOUT 45)
  endif()
  if(WIN32 AND INKHORN_WITH_PDFIUM)
    add_executable(pdf_tests tests/pdf_tests.cpp)
    target_link_libraries(pdf_tests PRIVATE inkhorn_pdf)
    add_test(NAME pdf_roundtrip COMMAND pdf_tests)
  endif()
endif()
